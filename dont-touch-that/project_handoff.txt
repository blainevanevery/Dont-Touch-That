--- LEVEL BUILDER GD ---
@tool
extends Node3D

@export var rebuild: bool = false:
	set(val):
		if val:
			build_level()

func _ready():
	build_level()

func build_level():
	# Clear existing children except Camera3D or default nodes
	for child in get_children():
		if child.name != "Camera3D" and child.name != "WorldEnvironment":
			child.queue_free()
			remove_child(child)

	# Read res://house_layout.json
	var file = FileAccess.open("res://house_layout.json", FileAccess.READ)
	if not file:
		push_error("Failed to open house_layout.json")
		return
	
	var json_text = file.get_as_text()
	var json = JSON.new()
	var err = json.parse(json_text)
	if err != OK:
		push_error("Failed to parse house_layout.json: " + json.get_error_message())
		return
		
	var data = json.get_data()
	var dimensions = data.get("dimensions", {})
	var yard_dims = dimensions.get("yard", [36.58, 13.72])
	var house_dims = dimensions.get("house", [18.29, 7.32])
	var wall_height = dimensions.get("wall_height", 3.0)
	var wall_thickness = dimensions.get("wall_thickness", 0.18)

	# 1. Create Yard (CSGBox3D)
	var yard = CSGBox3D.new()
	yard.name = "Yard"
	yard.use_collision = true
	yard.size = Vector3(yard_dims[1], 0.2, yard_dims[0]) # [length, width] -> width is X, length is Z
	yard.position = Vector3(0, -0.1, 0)
	var yard_mat = get_material("res://assets/materials/grass.tres", Color(0.15, 0.35, 0.15))
	yard.material = yard_mat
	add_child(yard)

	var wall_mat = get_material("res://assets/materials/wall.tres", Color(0.85, 0.85, 0.8))
	var floor_mat = get_material("res://assets/materials/wood.tres", Color(0.45, 0.3, 0.2))
	var roof_mat = get_material("res://assets/materials/roof.tres", Color(0.2, 0.2, 0.25))

	# Parse Floors
	var floors_data = data.get("floors", [])
	for floor_data in floors_data:
		var floor_num = floor_data.get("floor_number", 1)
		var y_base = (floor_num - 1) * wall_height
		
		# Create Floor plate (addition)
		var floor_combiner = CSGCombiner3D.new()
		floor_combiner.name = "House_Floor_" + str(floor_num)
		floor_combiner.use_collision = true
		add_child(floor_combiner)
		
		var floor_plate = CSGBox3D.new()
		floor_plate.name = "FloorPlate"
		floor_plate.size = Vector3(house_dims[1], 0.2, house_dims[0])
		floor_plate.position = Vector3(0, y_base + 0.1, 0)
		floor_plate.material = floor_mat
		floor_combiner.add_child(floor_plate)

		# Build Walls
		var walls = floor_data.get("walls", [])
		for w in walls:
			var start = w.get("start", [0.0, 0.0])
			var end = w.get("end", [0.0, 0.0])
			var thickness = w.get("thickness", wall_thickness)
			build_wall(Vector2(start[0], start[1]), Vector2(end[0], end[1]), y_base, wall_height, thickness, floor_combiner, wall_mat)
			
		# Build Doors (Subtractions)
		var doors = floor_data.get("doors", [])
		for d in doors:
			var dx = d.get("x", 0.0)
			var dz = d.get("z", 0.0)
			var dw = d.get("width", 0.9)
			var dh = d.get("height", 2.1)
			var rot = d.get("rotation", 0.0)
			
			var door_cut = CSGBox3D.new()
			door_cut.operation = CSGShape3D.OPERATION_SUBTRACTION
			door_cut.size = Vector3(dw, dh, 0.5)
			door_cut.position = Vector3(dx, y_base + dh / 2.0, dz)
			door_cut.rotation_degrees.y = rot
			floor_combiner.add_child(door_cut)

		# Build Windows (Subtractions)
		var windows = floor_data.get("windows", [])
		for win in windows:

# Wall builder helper
func build_wall(start: Vector2, end: Vector2, y_bottom: float, height: float, thickness: float, parent: Node, material: Material):
	var s3d = Vector3(start.x, 0, start.y)
	var e3d = Vector3(end.x, 0, end.y)
	var center = (s3d + e3d) / 2.0
	var dir = e3d - s3d
	var length = dir.length()
	if length < 0.01:
		return
		
	var box = CSGBox3D.new()
	box.use_collision = true
	box.material = material
	box.size = Vector3(thickness, height, length)
	box.position = Vector3(center.x, y_bottom + height / 2.0, center.z)
	
	var angle = atan2(dir.x, dir.z)
	box.rotation.y = angle
	parent.add_child(box)

# Stair builder helper
func build_stairs(x: float, z: float, width: float, length: float, y_start: float, y_end: float, direction: Array, parent: Node, material: Material):
	var num_steps = 15
	var step_depth = length / num_steps
	var step_height = (y_end - y_start) / num_steps
	var dx = direction[0]
	var dz = direction[1]
	
	for i in range(num_steps):
		var step = CSGBox3D.new()
		step.use_collision = true
		step.material = material
		step.size = Vector3(width, step_height * (i + 1), step_depth)
		
		# Offset position based on step index along direction
		var offset_l = (i * step_depth) + (step_depth / 2.0) - (length / 2.0)
		var step_x = x + dx * offset_l
		var step_z = z + dz * offset_l
		
		step.position = Vector3(step_x, y_start + (step_height * (i + 1)) / 2.0, step_z)
		parent.add_child(step)

# Helper to add wall
func add_wall_ft(x1_ft: float, z1_ft: float, x2_ft: float, z2_ft: float, y_bottom: float, y_top: float, thickness_m: float, parent: Node, material: Material):
	var w = CSGBox3D.new()
	w.use_collision = true
	w.material = material
	
	var x1 = get_x(x1_ft)
	var z1 = get_z(z1_ft)
	var x2 = get_x(x2_ft)
	var z2 = get_z(z2_ft)
	
	var dx = abs(x2 - x1)
	var dz = abs(z2 - z1)
	var dy = y_top - y_bottom
	
	if dx < 0.01: # Vertical wall (aligned with Z-axis)
		w.size = Vector3(thickness_m, dy, dz)
		w.position = Vector3(x1, y_bottom + dy/2.0, (z1 + z2)/2.0)
	elif dz < 0.01: # Horizontal wall (aligned with X-axis)
		w.size = Vector3(dx, dy, thickness_m)
		w.position = Vector3((x1 + x2)/2.0, y_bottom + dy/2.0, z1)
	else:
		w.size = Vector3(dx, dy, dz)
		w.position = Vector3((x1 + x2)/2.0, y_bottom + dy/2.0, (z1 + z2)/2.0)
		
	parent.add_child(w)

# Helper to build stairs
func build_stairs_ft(x1_ft: float, z1_ft: float, x2_ft: float, z2_ft: float, y_start: float, y_end: float, num_steps: int, parent: Node):
	var x1 = get_x(x1_ft)
	var z1 = get_z(z1_ft)
	var x2 = get_x(x2_ft)
	var z2 = get_z(z2_ft)
	
	var step_width = abs(x2 - x1)
	var total_run = abs(z2 - z1)
	var step_depth = total_run / num_steps
	var step_height = (y_end - y_start) / num_steps
	
	for i in range(num_steps):
		var step = CSGBox3D.new()
		step.use_collision = true
		step.size = Vector3(step_width, step_height * (i + 1), step_depth)
		
		# Direction logic (rising from front to back or back to front)
		var current_z = z2 - step_depth * i - step_depth / 2.0 if z1 < z2 else z2 + step_depth * i + step_depth / 2.0
		step.position = Vector3(
			(x1 + x2) / 2.0,
			y_start + (step_height * (i + 1)) / 2.0,
			current_z
		)
		step.material = get_wood_material()
		parent.add_child(step)
